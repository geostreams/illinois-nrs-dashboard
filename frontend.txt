'use client';
import React, { createContext, useContext, useState, useEffect, ReactNode } from "react";
import { useTheme, useMediaQuery } from "@mui/material";

interface ScreenSizeContextType {
    isSmallScreen: boolean;
}

// Create context with a default value matching the interface
const ScreenSizeContext = createContext<ScreenSizeContextType | undefined>(undefined);

interface ScreenSizeProviderProps {
    children: ReactNode;
}

export const ScreenSizeProvider: React.FC<ScreenSizeProviderProps> = ({ children }) => {
    const theme = useTheme();
    const isSmallScreen = useMediaQuery(theme.breakpoints.down("md"));
    const [screenSize, setScreenSize] = useState<ScreenSizeContextType>({
        isSmallScreen: isSmallScreen,
    });

    useEffect(() => {
        setScreenSize({ isSmallScreen: isSmallScreen });
    }, [isSmallScreen]);

    return (
        <ScreenSizeContext.Provider value={screenSize}>
            {children}
        </ScreenSizeContext.Provider>
    );
};

export const useScreenSize = (): ScreenSizeContextType => {
    const context = useContext(ScreenSizeContext);
    if (context === undefined) {
        throw new Error('useScreenSize must be used within a ScreenSizeProvider');
    }
    return context;
};

'use client';

import React from 'react';
import {
    PieChart,
    Pie,
    Cell,
    Tooltip,
    Legend,
    ResponsiveContainer,
} from "recharts";

// Define types for the data structure
type DataTuple = [string, number];

interface DataPoint {
    [key: string]: string | number;
    name: string;
    value: number;
}

interface Colors {
    [key: string]: string;
}

interface FilledPieChartProps {
    data: Array<DataPoint | DataTuple>;
    width: string | number;
    height: string | number;
    colors: Colors;
    fontSize?: number;
    innerRadius?: number | string;
    outerRadius?: number | string;
    dataKey?: string;
    nameKey?: string;
    legend?: boolean;
}

const FilledPieChart: React.FC<FilledPieChartProps> = ({
    data,
    width,
    height,
    colors,
    fontSize = 12,
    innerRadius = 0,
    outerRadius = "80%",
    dataKey = "value",
    nameKey = "name",
    legend = true,
}) => {
    // Function to get legend name, works with both object and array data
    const getLegendName = (entry: string | DataPoint): string => {
        if (typeof entry === 'string') {
            return entry
                .replace(/([A-Z])/g, " $1")
                .replace(/^./, (str) => str.toUpperCase());
        }

        const name = entry[nameKey];
        return typeof name === "string"
            ? name
                .replace(/([A-Z])/g, " $1")
                .replace(/^./, (str) => str.toUpperCase())
            : String(name);
    };

    // Prepare data for Pie component
    const pieData: DataPoint[] = Array.isArray(data[0])
        ? (data as DataTuple[]).map(([name, value]) => ({
            [nameKey]: name,
            [dataKey]: value,
            name, // ensure name property exists
            value // ensure value property exists
        }))
        : data as DataPoint[];

    return (
        <ResponsiveContainer width={width} height={height}>
            <PieChart>
                <Pie
                    data={pieData}
                    dataKey={dataKey}
                    nameKey={nameKey}
                    cx="50%"
                    cy="50%"
                    innerRadius={innerRadius}
                    outerRadius={outerRadius}
                    label
                >
                    {pieData.map((entry, index) => (
                        <Cell
                            key={`cell-${index}`}
                            fill={colors[entry.name]}
                        />
                    ))}
                </Pie>
                <Tooltip contentStyle={{ fontSize }} />
                {legend && (
                    <Legend
                        wrapperStyle={{ fontSize }}
                        formatter={(value: string) => getLegendName(value)}
                    />
                )}
            </PieChart>
        </ResponsiveContainer>
    );
};

export default FilledPieChart;

'use client';
import React from 'react';
import {
    BarChart,
    Bar,
    XAxis,
    YAxis,
    CartesianGrid,
    Tooltip,
    Legend,
    ResponsiveContainer,
    ReferenceLine
} from "recharts";

// Define types for the data structure
interface DataPoint {
    name: string;
    [key: string]: string | number; // Allow for dynamic data keys
}

// Define type for our custom reference line, extending Recharts' type
interface CustomReferenceLine {
    value: number;
    label: string;
}

// Define type for bar colors
interface BarColors {
    [key: string]: string;
}

// Define props interface
interface StackedBarChartProps {
    data: DataPoint[];
    width: string | number;
    height: string | number;
    barColors: BarColors;
    fontSize?: number;
    referenceLines?: CustomReferenceLine[];
    legend?: boolean;
}

const StackedBarChart: React.FC<StackedBarChartProps> = ({
    data,
    width,
    height,
    barColors,
    fontSize = 12,
    referenceLines,
    legend = true,
}) => {
    // Convert data keys to legend names
    const getLegendName = (key: string): string => {
        return key
            .replace(/([A-Z])/g, " $1")
            .replace(/^./, (str) => str.toUpperCase());
    };

    return (
        <ResponsiveContainer width={width} height={height}>
            <BarChart
                data={data}
                margin={{
                    top: 5,
                    right: 30,
                    left: 20,
                    bottom: 5,
                }}
            >
                <CartesianGrid strokeDasharray="3 3" />
                <XAxis dataKey="name" tick={{ fontSize }} />
                <YAxis tick={{ fontSize }} />
                <Tooltip contentStyle={{ fontSize }} />
                {legend && <Legend wrapperStyle={{ fontSize }} />}
                {data[0] &&
                    Object.keys(data[0])
                        .filter((key) => key !== "name")
                        .map((key) => (
                            <Bar
                                key={key}
                                dataKey={key}
                                stackId="a"
                                fill={barColors[key]}
                                name={getLegendName(key)}
                            />
                        ))}
                {referenceLines &&
                    referenceLines.map((line) => (
                        <ReferenceLine
                            key={line.value}
                            y={line.value}
                            label={{
                                value: line.label,
                                position: "insideTopRight",
                                fontSize,
                                color: "black",
                            }}
                            stroke="red"
                        />
                    ))}
            </BarChart>
        </ResponsiveContainer>
    );
};

export default StackedBarChart;

import React, { useState } from "react";
import { Grid, Typography, Link, Box } from "@mui/material";
import { useScreenSize } from "../context/ScreenSizeContext";

// Define types for the component props
type ComponentPosition = "top" | "bottom" | "left" | "right";
type GridSize = {
    xs: number;
    md: number;
};

interface TextDisplayProps {
    text: string;
    additionalComponent?: React.ReactNode;
    truncate?: boolean;
    componentPosition?: ComponentPosition;
    textComponentSize?: GridSize;
    propComponentSize?: GridSize;
}

const TextDisplay: React.FC<TextDisplayProps> = ({
    text,
    additionalComponent,
    componentPosition = "top",
    textComponentSize = { xs: 12, md: 5 },
    propComponentSize = { xs: 12, md: 7 },
}) => {
    const { isSmallScreen } = useScreenSize();


    const textContent = (
        <Box sx={{ mb: isSmallScreen ? 2 : 0 }}>
            <Typography variant="body1">
                {text}
            </Typography>
        </Box>
    );

    const additionalContent = additionalComponent && (
        <Box
            sx={{
                width: "100%",
                height: isSmallScreen ? 300 : 400,
                minHeight: 300,
            }}
        >
            {additionalComponent}
        </Box>
    );

    // Handle the layout if additional prop is not provided
    if (!additionalComponent) {
        return (
            <Grid
                container
                spacing={2}
                alignItems="stretch"
                direction={isSmallScreen ? "column" : "row"}
            >
                <Grid item xs={12}>
                    {textContent}
                </Grid>
            </Grid>
        );
    }

    switch (componentPosition) {
    case "top":
        return (
            <Grid
                container
                spacing={2}
                alignItems="stretch"
                direction="column"
            >
                <Grid item xs={12}>
                    {additionalContent}
                </Grid>
                <Grid item xs={12}>
                    {textContent}
                </Grid>
            </Grid>
        );
    case "bottom":
        return (
            <Grid
                container
                spacing={2}
                alignItems="stretch"
                direction="column"
            >
                <Grid item xs={12}>
                    {textContent}
                </Grid>
                <Grid item xs={12}>
                    {additionalContent}
                </Grid>
            </Grid>
        );

    case "left":
        return (
            <Grid
                container
                spacing={2}
                alignItems="stretch"
                direction={isSmallScreen ? "column" : "row"}
            >
                <Grid
                    item
                    xs={propComponentSize.xs}
                    md={propComponentSize.md}
                >
                    {additionalContent}
                </Grid>
                <Grid
                    item
                    xs={textComponentSize.xs}
                    md={textComponentSize.md}
                >
                    {textContent}
                </Grid>
            </Grid>
        );
    case "right":
    default:
        return (
            <Grid
                container
                spacing={2}
                alignItems="stretch"
                direction={isSmallScreen ? "column" : "row"}
            >
                <Grid
                    item
                    xs={textComponentSize.xs}
                    md={textComponentSize.md}
                >
                    {textContent}
                </Grid>
                <Grid
                    item
                    xs={propComponentSize.xs}
                    md={propComponentSize.md}
                >
                    {additionalContent}
                </Grid>
            </Grid>
        );
    }
};

export default TextDisplay;
